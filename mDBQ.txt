// ============================================================
// MongoDB Queries – moviesDB.moviesColl
// Database : moviesDB
// Collection: moviesColl
// ============================================================

// ── Query 1 ──────────────────────────────────────────────────
// Create an index on the 'name' field for fast lookups, then
// search for a specific title by exact name.
use moviesDB

db.moviesColl.createIndex({ name: 1 })

db.moviesColl.findOne({ name: "Ice Airport Alaska" })


// ── Query 2 ──────────────────────────────────────────────────
// Find all titles whose type is "Movie" AND whose genre field
// contains either "Drama" or "Documentary" (genre may hold a
// comma-separated list, so $regex is used for a substring match).
use moviesDB

db.moviesColl.find({
  type: "Movie",
  $or: [
    { genre: { $regex: "Drama",       $options: "i" } },
    { genre: { $regex: "Documentary", $options: "i" } }
  ]
})


// ── Query 3 ──────────────────────────────────────────────────
// Aggregation pipeline: count how many titles exist per type
// (e.g., "Movie" vs "TV Show"), sorted from highest to lowest.
use moviesDB

db.moviesColl.aggregate([
  {
    $group: {
      _id:   "$type",
      count: { $sum: 1 }
    }
  },
  {
    $sort: { count: -1 }
  }
])


// ── Query 4 ──────────────────────────────────────────────────
// Find the top 5 highest-rated Movies released after 2010 that
// are available on "Prime Video" or "Netflix" and have a non-
// empty imdb_rating.  The rating is stored as "7.8/10" so we
// use $addFields + $toDouble on the numeric prefix for sorting.
use moviesDB

db.moviesColl.aggregate([
  // Keep only Movies with a rating, released after 2010, on
  // one of the two platforms.
  {
    $match: {
      type:        "Movie",
      imdb_rating: { $exists: true, $nin: ["", null] },
      released_at: { $gt: "2010-12-31" },
      $or: [
        { streaming_on: { $regex: "Prime Video", $options: "i" } },
        { streaming_on: { $regex: "Netflix",     $options: "i" } }
      ]
    }
  },
  // Extract the numeric part before the "/" so we can sort numerically.
  {
    $addFields: {
      rating_numeric: {
        $toDouble: {
          $arrayElemAt: [ { $split: [ "$imdb_rating", "/" ] }, 0 ]
        }
      }
    }
  },
  // Sort by the numeric rating descending.
  { $sort: { rating_numeric: -1 } },
  // Keep only the top 5.
  { $limit: 5 },
  // Project only the required fields.
  {
    $project: {
      _id:          0,
      name:         1,
      released_at:  1,
      genre:        1,
      streaming_on: 1,
      country:      1,
      imdb_rating:  1
    }
  }
])


// ── Query 5 (Bonus) ──────────────────────────────────────────
// Advanced query using $facet to produce a multi-dimensional
// summary in a single pass:
//   • "by_genre"   – top 5 genres by title count (genres are
//                    split from the comma-separated string)
//   • "rating_stats" – average, min, and max IMDb rating across
//                      all documents that have a valid rating
//   • "by_content_rating" – count of titles per content rating,
//                            sorted descending
use moviesDB

db.moviesColl.aggregate([
  // ── shared pre-processing ──────────────────────────────────
  // Convert "7.8/10" → 7.8  (null when the field is absent/empty)
  {
    $addFields: {
      rating_numeric: {
        $cond: {
          if: { $and: [
            { $gt: [ "$imdb_rating", null ] },
            { $ne:  [ "$imdb_rating", ""  ] }
          ]},
          then: {
            $toDouble: {
              $arrayElemAt: [ { $split: [ "$imdb_rating", "/" ] }, 0 ]
            }
          },
          else: null
        }
      },
      // Split "Drama, Documentary" → ["Drama", " Documentary"]
      genre_array: {
        $map: {
          input: { $split: [ "$genre", "," ] },
          as:   "g",
          in:   { $trim: { input: "$$g" } }
        }
      }
    }
  },

  // ── $facet: run three sub-pipelines in parallel ────────────
  {
    $facet: {

      // 1. Top 5 genres by number of titles
      "by_genre": [
        { $unwind: "$genre_array" },
        { $group:  { _id: "$genre_array", count: { $sum: 1 } } },
        { $sort:   { count: -1 } },
        { $limit:  5 },
        { $project: { _id: 0, genre: "$_id", count: 1 } }
      ],

      // 2. IMDb rating statistics (only docs with a valid rating)
      "rating_stats": [
        { $match: { rating_numeric: { $ne: null } } },
        {
          $group: {
            _id:        null,
            avg_rating: { $avg: "$rating_numeric" },
            max_rating: { $max: "$rating_numeric" },
            min_rating: { $min: "$rating_numeric" },
            total_rated: { $sum: 1 }
          }
        },
        { $project: { _id: 0, avg_rating: { $round: ["$avg_rating", 2] },
                      max_rating: 1, min_rating: 1, total_rated: 1 } }
      ],

      // 3. Title count per content rating, descending
      "by_content_rating": [
        { $match: { content_rating: { $exists: true, $nin: ["", null] } } },
        { $group:  { _id: "$content_rating", count: { $sum: 1 } } },
        { $sort:   { count: -1 } },
        { $project: { _id: 0, content_rating: "$_id", count: 1 } }
      ]
    }
  }
])
